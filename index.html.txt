                // Clear any previous error messages if initialization starts
                if (firebaseErrorMessageDisplay) {
                    firebaseErrorMessageDisplay.classList.add('hidden');
                    firebaseErrorMessageDisplay.textContent = '';
                }

                // Explicitly check for an empty Firebase config or missing projectId

                        firebaseErrorMessageDisplay.classList.remove('hidden');
                    }
                    return; // Stop initialization if config is empty
                }

                if (!firebaseConfig.projectId) {
                    console.error("Firebase 'projectId' is missing in the configuration.");
                    if (firebaseErrorMessageDisplay) {
                        firebaseErrorMessageDisplay.textContent = "Firebase project ID is missing from the configuration. Please ensure your Canvas environment provides a valid 'projectId'.";
                        firebaseErrorMessageDisplay.classList.remove('hidden');
                    }
                    return; // Stop initialization if projectId is missing
                }


                const app = initializeApp(firebaseConfig);
                db = getFirestore(app);
                auth = getAuth(app);


                // Sign in using custom token or anonymously if token is not available
                if (initialAuthToken) {
                    await signInWithCustomToken(auth, initialAuthToken);
                    console.log("Signed in with custom token.");
                } else {
                    await signInAnonymously(auth);
                    console.log("Signed in anonymously.");
                }

                // Listen for auth state changes to get the current user ID
                onAuthStateChanged(auth, (user) => {
                    if (user) {

                    } else {
                        // User is signed out. For anonymous, a new ID might be generated on next session.
                        console.log("User signed out or no user.");
                        userId = null; // Clear userId if signed out
                    }
                });

            } catch (error) {
                console.error("Error initializing Firebase or signing in:", error);
                if (firebaseErrorMessageDisplay) {
                    if (error.code === 'auth/network-request-failed') {
                        firebaseErrorMessageDisplay.textContent = "Network error connecting to our services. Please check your internet connection or try refreshing the page.";
                    } else {
                        firebaseErrorMessageDisplay.textContent = `Error initializing features: ${error.message}. Please try refreshing the page.`;
                    }

        // --- GENERAL HELPER FOR GEMINI API CALLS ---
        async function callGeminiAPI(prompt, chatHistory, outputElement, errorElement, loadingSpinner, buttonTextElement) {
            if (!prompt) return; // Ensure there's a prompt

            // Add the current user prompt to history
            chatHistory.push({ role: "user", parts: [{ text: prompt }] });

            const payload = { contents: chatHistory };
            const apiKey = ""; // Canvas will automatically provide this at runtime.
                               // IMPORTANT: User needs to ensure this is correctly handled by their environment.
            if (!apiKey) {
                console.error("Gemini API Key is missing. Please ensure it's configured in your Canvas environment.");
                if (errorElement) {
                    errorElement.classList.remove('hidden');
                    errorElement.innerHTML = '<p>AI service not available. API Key missing.</p>';
                }
                if (outputElement) outputElement.parentElement.classList.add('hidden');
                return; // Exit if API key is missing
            }


            const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-05-20:generateContent?key=${apiKey}`;

            const originalButtonText = buttonTextElement.dataset.originalText || buttonTextElement.textContent;
            buttonTextElement.textContent = 'Processing...';
            loadingSpinner.classList.remove('hidden');
            buttonTextElement.parentElement.disabled = true;
            outputElement.parentElement.classList.add('hidden'); // Hide result container
            errorElement.classList.add('hidden'); // Hide error message

            const maxRetries = 5;
            let currentRetry = 0;
            let success = false;

            while (currentRetry < maxRetries && !success) {
                try {
                    const response = await fetch(apiUrl, {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify(payload)
                    });

                    if (!response.ok) {
                        const errorData = await response.json();
                        console.error("API Error Response:", errorData);
                        throw new Error(`HTTP error! status: ${response.status} - ${errorData.error?.message || 'Unknown error'}`);
                    }

                    const result = await response.json();

                    if (result.candidates && result.candidates.length > 0 &&
                        result.candidates[0].content && result.candidates[0].content.parts &&
                        result.candidates[0].content.parts.length > 0) {
                        const text = result.candidates[0].content.parts[0].text;
                        outputElement.innerHTML = marked.parse(text); // Use innerHTML to allow basic formatting from LLM
                        outputElement.parentElement.classList.remove('hidden'); // Show the result container
                        chatHistory.push({ role: "model", parts: [{ text: text }] }); // Add model's response to history
                        success = true;
                    } else {
                        throw new Error('Unexpected API response structure or no content.');
                    }
                } catch (error) {
                    console.error('Error fetching recommendation:', error);
                    currentRetry++;
                    if (currentRetry < maxRetries) {
                        const delay = Math.pow(2, currentRetry) * 1000; // Exponential backoff
                        await new Promise(res => setTimeout(res, delay));
                    } else {
                        errorElement.classList.remove('hidden'); // Show error message
                        errorElement.innerHTML = `<p>Error: ${error.message}. Please try again.</p>`;
                        outputElement.innerHTML = '';
                    }
                } finally {
                    if (currentRetry >= maxRetries || success) {
                        buttonTextElement.textContent = originalButtonText; // Restore original text
                        loadingSpinner.classList.add('hidden');
                        buttonTextElement.parentElement.disabled = false;
                    }
                }
            }
        }

            // No need to initialize waver-profile-form as it's removed.

            // Ask the Community Guru AI
            const askAiBtn = document.getElementById('ask-ai-btn');
            const aiQuestionInput = document.getElementById('ai-question');
            const aiResponseContainer = document.getElementById('ai-response-container');
            const aiResponseText = document.getElementById('ai-response');
            const aiLoadingIndicator = document.getElementById('ai-loading-indicator');
            const askAiButtonText = askAiBtn ? (askAiBtn.querySelector('span') || askAiBtn) : null; // Handle if button itself is null

            if (askAiBtn && aiQuestionInput && aiResponseContainer && aiResponseText && aiLoadingIndicator && askAiButtonText) { // Ensure all elements exist
                askAiButtonText.dataset.originalText = askAiButtonText.textContent;

                askAiBtn.addEventListener('click', async () => {
                    const userQuestion = aiQuestionInput.value.trim();
                    if (!userQuestion) {
                        if (aiResponseContainer) {
                             aiResponseContainer.classList.remove('hidden');
                             if (aiResponseText) aiResponseText.innerHTML = '<p class="text-red-400">Please ask a question.</p>';
                        }
                        return;
                    }

                    // Prepend system instruction to the current question for context
                    const fullPrompt = `As the Ask Guru AI, answer the following question about 360 waves or wave care. Be helpful, concise, and informative. Incorporate general tips that might benefit a waver.

Question: "${userQuestion}"`;

                    await callGeminiAPI(fullPrompt, communityChatHistory, aiResponseText, document.getElementById('ai-response-container').nextElementSibling, aiLoadingIndicator, askAiButtonText);
                    aiQuestionInput.value = ''; // Clear input after sending
                });
            } else {
                console.error("Ask the Community Guru elements not found correctly.");
            }

            // Close modal when clicking outside of it (assuming waverDetailsModal exists)
            const waverDetailsModal = document.getElementById('waverDetailsModal');
            if (waverDetailsModal) {
                waverDetailsModal.addEventListener('click', (e) => {
                    if (e.target.id === 'waverDetailsModal') {
                        hideWaverDetails();
                    }
                });
            }

        }); // End of consolidated DOMContentLoaded listener

        // JavaScript for mobile menu toggle functionality (globally accessible due to onclick)
        function toggleMobileMenu() {
            const mobileMenu = document.getElementById('mobile-menu');
            if (mobileMenu) mobileMenu.classList.toggle('hidden');
        }

        // Smooth scrolling for all internal navigation links (those starting with '#')
        document.querySelectorAll('a[href^="#"]').forEach(anchor => {
            anchor.addEventListener('click', function (e) {
                e.preventDefault();
                const targetElement = document.querySelector(this.getAttribute('href'));
                if (targetElement) {
                    targetElement.scrollIntoView({
                        behavior: 'smooth'
                    });
                }
                // Close mobile menu after clicking a link, if it's open
                const mobileMenu = document.getElementById('mobile-menu');
                if (mobileMenu && !mobileMenu.classList.contains('hidden')) {
                    toggleMobileMenu();
                }
            });
        });
